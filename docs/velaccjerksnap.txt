// =============================================================================
// 2. 物理演算 & 正規化
// =============================================================================
f_hma(src, len) => ta.wma(2 * ta.wma(src, int(len/2)) - ta.wma(src, len), int(math.round(math.sqrt(len))))
f_norm(src, len) =>
    float dev = nz(ta.stdev(src, len))
    dev != 0 ? src / dev : 0

float src_smooth = f_hma(close, len_smooth)
float mid_line   = f_hma(src_smooth, len_hma)

float raw_vel  = mid_line - mid_line[1]
float raw_acc  = raw_vel - raw_vel[1]
float raw_jerk = raw_acc - raw_acc[1]
float raw_snap = raw_jerk - raw_jerk[1]

float n_vel  = f_norm(raw_vel,  len_norm)
float n_acc  = f_norm(raw_acc,  len_norm)
float n_jerk = f_norm(raw_jerk, len_norm)
float n_snap = f_norm(raw_snap, len_norm)


// =============================================================================
// 3. マスク演算 (Mask Logic) & 追加フィルター
// =============================================================================
f_elastic_system(float src_ma, float src_raw, int len_med, float coef, float rope, int smooth) =>
    float body_size = math.abs(close - open)
    float median_body = ta.median(body_size, len_med)
    float band_width = median_body * coef
    
    var float drag_point = na
    if na(drag_point)
        drag_point := src_ma
    
    float slack = band_width * rope
    if src_ma > drag_point + slack
        drag_point := src_ma - slack
    else if src_ma < drag_point - slack
        drag_point := src_ma + slack
    
    float ma_inertia = ta.sma(drag_point, smooth)
    
    // ステートマシン
    var int state = 0
    var float anchor = na
    var float limit = na
    
    if not na(ma_inertia)
        if na(anchor)
            anchor := ma_inertia
            limit  := ma_inertia - band_width
            state := 1
        else
            if state == 1 // Bull
                if ma_inertia > anchor
                    anchor := ma_inertia
                limit := anchor - band_width
                
                if ma_inertia < limit
                    state := -1
                    anchor := ma_inertia
                    limit := anchor + band_width
            else // Bear
                if ma_inertia < anchor
                    anchor := ma_inertia
                limit := anchor + band_width
                
                if ma_inertia > limit
                    state := 1
                    anchor := ma_inertia
                    limit := anchor - band_width

    [state]

// マスク用MA計算 (ソース選択)
float custom_src = switch mask_src_type
    "Close" => close
    "Open" => open
    "High" => high
    "Low" => low
    "BodyMid (Open+Close)/2" => (open + close) / 2
    => close

float mask_ma = ta.sma(custom_src, mask_len_ma)

// 計算実行
[mask_state] = f_elastic_system(mask_ma, mask_ma, mask_len_median, mask_width_coef, mask_rope, mask_smooth)

// --- 追加フィルター計算 ---
// 1. MA Filter
float val_hma72 = ta.wma(2*ta.wma(close, 36) - ta.wma(close, 72), 8)
float val_sma72 = ta.sma(close, 72)
bool ma_trend_down = val_hma72 < val_hma72[1] and val_sma72 < val_sma72[1]
bool ma_trend_up   = val_hma72 > val_hma72[1] and val_sma72 > val_sma72[1]

// 2. Vel Filter (Dead Zone)
bool vel_is_weak = math.abs(n_vel) < opt_vel_thresh

// --- 統合判定 (Total Check) ---
// Longに対する判定
// 禁止条件: (マスクがBearかつ見送り) OR (MA禁止かつ下降) OR (Vel禁止かつ弱)
bool is_long_ban = (use_trend_mask and mask_state == -1 and mask_action == "見送り") or (opt_ma_type == "禁止 (Ban)" and ma_trend_down) or (opt_vel_type == "禁止 (Ban)" and vel_is_weak)

// 待機条件: 禁止ではなく、((マスクがBearかつ待機) OR (MA待機かつ下降) OR (Vel待機かつ弱))
bool is_long_wait_cond = (use_trend_mask and mask_state == -1 and mask_action == "待機") or (opt_ma_type == "待機 (Wait)" and ma_trend_down) or (opt_vel_type == "待機 (Wait)" and vel_is_weak)
bool is_long_wait = not is_long_ban and is_long_wait_cond

// Shortに対する判定
bool is_short_ban = (use_trend_mask and mask_state == 1 and mask_action == "見送り") or (opt_ma_type == "禁止 (Ban)" and ma_trend_up) or (opt_vel_type == "禁止 (Ban)" and vel_is_weak)

bool is_short_wait_cond = (use_trend_mask and mask_state == 1 and mask_action == "待機") or (opt_ma_type == "待機 (Wait)" and ma_trend_up) or (opt_vel_type == "待機 (Wait)" and vel_is_weak)
bool is_short_wait = not is_short_ban and is_short_wait_cond


// =============================================================================
// 4. 条件判定ロジック
// =============================================================================
check_layer_mirror(val, val_1, ref_val, slp_opt, pol_opt, hier_opt, is_inverse) =>
    string target_slp = slp_opt
    if is_inverse and slp_opt != "無視"
        target_slp := slp_opt == "上昇" ? "下降" : "上昇"

    bool ok_slp = true
    if target_slp == "上昇"
        ok_slp := val > val_1
    else if target_slp == "下降"
        ok_slp := val < val_1
    
    string target_pol = pol_opt
    if is_inverse and pol_opt != "無視"
        target_pol := pol_opt == "正(+)" ? "負(-)" : "正(+)"
        
    bool ok_pol = true
    if target_pol == "正(+)"
        ok_pol := val > 0
    else if target_pol == "負(-)"
        ok_pol := val < 0
    
    string target_hier = hier_opt
    if is_inverse and hier_opt != "無視"
        if str.contains(hier_opt, "より上")
            target_hier := str.replace(hier_opt, "より上", "より下")
        else if str.contains(hier_opt, "より下")
            target_hier := str.replace(hier_opt, "より下", "より上")

    bool ok_hier = true
    if str.contains(target_hier, "より上")
        ok_hier := val > ref_val
    else if str.contains(target_hier, "より下")
        ok_hier := val < ref_val
        
    ok_slp and ok_pol and ok_hier

f_check_entry(is_inv) =>
    bool o_v = check_layer_mirror(n_vel, n_vel[1], 0.0, in_v_slp, in_v_pol, "無視", is_inv)
    bool o_a = check_layer_mirror(n_acc, n_acc[1], n_vel, in_a_slp, in_a_pol, in_a_hier, is_inv)
    bool o_j = check_layer_mirror(n_jerk, n_jerk[1], n_acc, in_j_slp, in_j_pol, in_j_hier, is_inv)
    bool o_s = check_layer_mirror(n_snap, n_snap[1], n_jerk, in_s_slp, in_s_pol, in_s_hier, is_inv)
    o_v and o_a and o_j and o_s

f_check_exit(is_inv) =>
    if not use_exit_logic
        false
    else
        bool o_v = check_layer_mirror(n_vel, n_vel[1], 0.0, out_v_slp, out_v_pol, "無視", is_inv)
        bool o_a = check_layer_mirror(n_acc, n_acc[1], n_vel, out_a_slp, out_a_pol, out_a_hier, is_inv)
        bool o_j = check_layer_mirror(n_jerk, n_jerk[1], n_acc, out_j_slp, out_j_pol, out_j_hier, is_inv)
        bool o_s = check_layer_mirror(n_snap, n_snap[1], n_jerk, out_s_slp, out_s_pol, out_s_hier, is_inv)
        o_v and o_a and o_j and o_s

bool entry_long  = f_check_entry(false) 
bool entry_short = f_check_entry(true)  
bool exit_long   = f_check_exit(false)  
bool exit_short  = f_check_exit(true)   
bool force_long  = use_force_entry and (n_vel >= force_vel_thresh)
bool force_short = use_force_entry and (n_vel <= -force_vel_thresh)


// =============================================================================
// 5. 売買実行 (Filters Integrated)
// =============================================================================
bool can_execute = (not wait_for_close) or barstate.isconfirmed

var bool pending_long = false
var bool pending_short = false

// 反対シグナルで待機キャンセル
if entry_short or force_short
    pending_long := false
if entry_long or force_long
    pending_short := false

// 待機中のBan条件発生チェック（待機中に禁止条件が成立したらキャンセル）
if pending_long and is_long_ban
    pending_long := false
if pending_short and is_short_ban
    pending_short := false


if can_execute
    // --- 0. MA逆行 & Vel決済オプション ---
    if opt_ma_exit
        if strategy.position_size > 0 and ma_trend_down
            strategy.close("Long", comment="Exit(MA)")
            pending_long := false
        if strategy.position_size < 0 and ma_trend_up
            strategy.close("Short", comment="Exit(MA)")
            pending_short := false
            
    // Vel範囲内での決済許可 (新規追加)
    // エントリーはブロックされるが、反対ポジションを持っていれば閉じる
    if opt_vel_exit_allow and vel_is_weak
        if strategy.position_size > 0 and (entry_short or force_short)
            strategy.close("Long", comment="Exit(Vel)")
        if strategy.position_size < 0 and (entry_long or force_long)
            strategy.close("Short", comment="Exit(Vel)")

    // --- A. 強制ドテン (Force Entry) ---
    if force_long
        if not is_long_ban and not is_long_wait
            strategy.entry("Long", strategy.long, comment="Force L")
            pending_long := false
        else if is_long_wait
            pending_long := true 
            
    else if force_short
        if not is_short_ban and not is_short_wait
            strategy.entry("Short", strategy.short, comment="Force S")
            pending_short := false
        else if is_short_wait
            pending_short := true
    
    // --- B. 通常エントリー (Normal Entry) ---
    else
        if entry_long
            if not is_long_ban and not is_long_wait
                strategy.entry("Long", strategy.long, comment="Entry L")
                pending_long := false
            else if is_long_wait
                pending_long := true
        
        if entry_short
            if not is_short_ban and not is_short_wait
                strategy.entry("Short", strategy.short, comment="Entry S")
                pending_short := false
            else if is_short_wait
                pending_short := true

    // --- C. 待機発動 (Wait Trigger) ---
    // すべての禁止/待機条件が解除されたら発動
    if pending_long and not is_long_ban and not is_long_wait
        strategy.entry("Long", strategy.long, comment="Wait L")
        pending_long := false
        
    if pending_short and not is_short_ban and not is_short_wait
        strategy.entry("Short", strategy.short, comment="Wait S")
        pending_short := false

    // --- D. カスタム決済 ---
    if strategy.position_size > 0 and exit_long and not entry_short and not pending_short
        strategy.close("Long", comment="Exit L")
        
    if strategy.position_size < 0 and exit_short and not entry_long and not pending_long
        strategy.close("Short", comment="Exit S")
